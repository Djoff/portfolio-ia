" | strings.TrimSpace -}} {{- with index $titres (lower $titre) -}} {{- $res = merge $res (dict . $corps) -}} {{- else -}} {{- $inconnues = $inconnues | append $titre -}} {{- end -}} {{- end -}} {{- return merge $res (dict "inconnues" $inconnues) -}}
